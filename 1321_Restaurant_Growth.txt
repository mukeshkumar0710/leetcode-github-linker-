WITH DailySales AS (
    SELECT 
        visited_on,
        SUM(amount) AS amount
    FROM 
        Customer
    GROUP BY 
        visited_on
),
RollingMetrics AS (
    SELECT 
        visited_on,
        SUM(amount) OVER (
            ORDER BY visited_on 
            ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
        ) AS amount,
        ROUND(AVG(amount) OVER (
            ORDER BY visited_on 
            ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
        ), 2) AS average_amount,
        ROW_NUMBER() OVER (ORDER BY visited_on) AS rn
    FROM 
        DailySales
)
SELECT 
    visited_on,
    amount,
    average_amount
FROM 
    RollingMetrics
WHERE 
    rn >= 7
ORDER BY 
    visited_on ASC;
